FROM maven:3.9-eclipse-temurin-17 AS build
WORKDIR /app
COPY . .
RUN mvn -f JavaNetbeans/pom.xml clean package -DskipTests

FROM tomcat:10.1-jdk17-temurin
RUN rm -rf /usr/local/tomcat/webapps/*
COPY --from=build /app/JavaNetbeans/target/ladonita.war /usr/local/tomcat/webapps/ROOT.war
ENV JAVA_OPTS="-Xmx300m"
EXPOSE 8080
